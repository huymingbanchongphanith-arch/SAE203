    CATEGORIE {
        int id PK
        varchar nom

    }

    PRODUIT {
        int id PK
        varchar nom
        date date_peremption
        varchar photo
        varchar marque

    }

    CLIENT {
        int id PK
        varchar nom
        varchar prenom
        date date_inscription
        varchar adresse
    }

    COMMANDE {
        int numero PK
        int client_id FK
        datetime date_commande
    }

        int commande_numero PK,FK
